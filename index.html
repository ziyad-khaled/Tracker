-- Weekly avg (Sat-Fri, div 7)
SELECT date_trunc('week', session_date + interval '2 days') - interval '2 days' AS week_sat,
  ROUND((SUM(focus_sec)/7.0)::NUMERIC,1) AS avg_per_day_min
FROM focus_sessions GROUP BY week_sat ORDER BY week_sat DESC LIMIT 4;

-- Break return rate by activity
SELECT break_activities, COUNT(*) AS n,
  ROUND(AVG(CASE WHEN returned THEN 1 ELSE 0 END)::NUMERIC*100,1) AS return_pct
FROM breaks WHERE break_activities IS NOT NULL
GROUP BY break_activities ORDER BY n DESC;

-- Sleep vs focus (join tables)
SELECT ROUND(dc.sleep_hrs::NUMERIC,0) AS sleep_hrs,
  COUNT(*) AS sessions, ROUND(AVG(fs.focus_sec)::NUMERIC,1) AS avg_focus
FROM focus_sessions fs
JOIN daily_checkins dc ON dc.date = fs.session_date
WHERE dc.sleep_hrs IS NOT NULL GROUP BY sleep_hrs ORDER BY sleep_hrs;

-- ║  focus_min → focus_sec  (run steps A1–A5 first)             ║

-- A1 — fill focus_sec where NULL
UPDATE focus_sessions
SET focus_sec = focus_min * 60
WHERE focus_sec IS NULL AND focus_min IS NOT NULL;

-- A2 — audit drift (hand-edited focus_min rows); review before touching
SELECT id, session_date, start_time,
  focus_min                              AS focus_min_stored,
  focus_sec                              AS focus_sec_stored,
  ROUND(focus_sec / 60.0, 2)             AS focus_sec_as_min,
  focus_min - ROUND(focus_sec / 60.0, 0) AS drift_min
FROM focus_sessions
WHERE focus_sec IS NOT NULL AND focus_min IS NOT NULL
  AND ABS(focus_min - ROUND(focus_sec / 60.0, 0)) > 1
ORDER BY ABS(focus_min - ROUND(focus_sec / 60.0, 0)) DESC;

-- A3 — fix drifted rows: trust focus_min (your manual edits)
UPDATE focus_sessions
SET focus_sec = focus_min * 60
WHERE focus_sec IS NOT NULL AND focus_min IS NOT NULL
  AND ABS(focus_min - ROUND(focus_sec / 60.0, 0)) > 1;

-- A4 — verify (should return 0)
SELECT COUNT(*) AS out_of_sync FROM focus_sessions
WHERE focus_sec IS NOT NULL AND focus_min IS NOT NULL
  AND ABS(focus_min - ROUND(focus_sec / 60.0, 0)) > 1;

-- A5 — drop focus_min — IRREVERSIBLE, run only after A4 = 0
ALTER TABLE focus_sessions DROP COLUMN focus_min;


-- ╔═══════════════════════════════════════════════════════════════╗
-- ║  span_min → span_sec  (run steps B1–B5 after A is done)     ║
-- ╚═══════════════════════════════════════════════════════════════╝

-- B1 — add span_sec column (seconds, derived from end_time - start_time)
ALTER TABLE focus_sessions ADD COLUMN span_sec int4;

-- B2 — populate span_sec from the actual timestamps
UPDATE focus_sessions
SET span_sec = EXTRACT(EPOCH FROM (
    (session_date + end_time::interval) -
    (session_date + start_time::interval)
  ))::int
WHERE start_time IS NOT NULL AND end_time IS NOT NULL;

-- B3 — audit: compare new span_sec to old span_min
-- Shows rows where they differ by more than 60 seconds (1 min)
SELECT id, session_date, start_time, end_time,
  span_min,
  span_sec,
  ROUND(span_sec / 60.0, 2)             AS span_sec_as_min,
  span_min - ROUND(span_sec / 60.0, 0)  AS drift_min
FROM focus_sessions
WHERE span_sec IS NOT NULL AND span_min IS NOT NULL
  AND ABS(span_min - ROUND(span_sec / 60.0, 0)) > 1
ORDER BY ABS(span_min - ROUND(span_sec / 60.0, 0)) DESC;

-- B4 — verify everything looks clean (should return 0 after review)
SELECT COUNT(*) AS out_of_sync FROM focus_sessions
WHERE span_sec IS NOT NULL AND span_min IS NOT NULL
  AND ABS(span_min - ROUND(span_sec / 60.0, 0)) > 1;

-- B5 — drop span_min — IRREVERSIBLE, run only after B4 = 0
ALTER TABLE focus_sessions DROP COLUMN span_min;
