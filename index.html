-- daily focus totals + avg ratio
SELECT session_date,
       SUM(focus_min)                         AS total_focus_min,
       ROUND(AVG(ratio)::NUMERIC, 1)          AS avg_ratio,
       COUNT(*)                               AS sessions
FROM sessions
GROUP BY session_date
ORDER BY session_date DESC LIMIT 14;

-- which break activity kills return rate the most
SELECT break_activities,
       COUNT(*)                               AS sessions,
       ROUND(AVG(CASE WHEN returned THEN 1 ELSE 0 END)::NUMERIC * 100, 1) AS return_pct,
       ROUND(AVG(break_duration_min)::NUMERIC, 1) AS avg_break_min
FROM sessions
WHERE break_activities IS NOT NULL AND break_activities != '—'
GROUP BY break_activities
ORDER BY sessions DESC;

-- energy level vs focus output
SELECT energy,
       COUNT(*)                               AS sessions,
       ROUND(AVG(focus_min)::NUMERIC, 1)      AS avg_focus,
       ROUND(AVG(ratio)::NUMERIC, 1)          AS avg_ratio
FROM sessions
WHERE energy IS NOT NULL
GROUP BY energy ORDER BY energy;

-- sleep hours vs next-day performance
SELECT ROUND(sleep_hrs::NUMERIC, 0) AS sleep_bucket,
       COUNT(*)                               AS sessions,
       ROUND(AVG(focus_min)::NUMERIC, 1)      AS avg_focus,
       ROUND(AVG(ratio)::NUMERIC, 1)          AS avg_ratio
FROM sessions
WHERE sleep_hrs IS NOT NULL
GROUP BY sleep_bucket ORDER BY sleep_bucket;

Table Setup — run once in Supabase SQL Editor
CREATE TABLE IF NOT EXISTS sessions (
  id                 UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at         TIMESTAMPTZ DEFAULT now(),
  session_date       DATE NOT NULL,
  start_time         TIME NOT NULL,
  end_time           TIME NOT NULL,
  span_min           INTEGER NOT NULL,
  focus_min          INTEGER NOT NULL,
  ratio              INTEGER,
  task               TEXT,
  task_type          TEXT,
  seq                INTEGER,
  energy             INTEGER,
  sleep_hrs          NUMERIC(3,1),
  wake_time          TIME,
  break_duration_min INTEGER,
  break_activities   TEXT,
  overdue            BOOLEAN DEFAULT false,
  returned           BOOLEAN,
  note               TEXT,
  break_note         TEXT
);

ALTER TABLE sessions ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Enable insert for anon" ON sessions FOR INSERT WITH CHECK (true);
CREATE POLICY "Enable select for anon" ON sessions FOR SELECT USING (true);
